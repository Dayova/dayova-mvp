import './global.css';
import { registerRootComponent } from 'expo';
import { useFonts } from 'expo-font';
import { useEffect, useState } from 'react';
import { Linking, View, ScrollView } from 'react-native';
import { DayovaThemeProvider } from './lib/theme';
import { Text } from './components/ui/text';
import { SelectionOptionRow } from './components/ui/selection-option-row';
import { Pencil } from './components/ui/icon';
import { ChoiceList } from './features/learning-plans/choice-list';
import { FeedbackView } from './features/learning-plans/session-feedback';

import { TextAnswer } from './features/learning-plans/text-answer';
import { Button } from './components/ui/button';
function Preview() {
 const [page,setPage]=useState(0); const [selected,setSelected]=useState('b');
 const [value,setValue]=useState('');
 const [loaded]=useFonts({Poppins:require('../assets/fonts/Poppins-Regular.ttf'),'Poppins-SemiBold':require('../assets/fonts/Poppins-SemiBold.ttf')});
 useEffect(()=>{const sub=Linking.addEventListener('url',({url})=>{const m=url.match(/screen=(\d)/);if(m)setPage(Number(m[1]));});return ()=>sub.remove();},[]);
 if(!loaded)return null;
 return <View className="flex-1 bg-background px-6 pt-20 pb-10">
 <Text className="mb-6 font-semibold text-heading-2 text-text">{['Prüfung eintragen','Schulmaterial hinzufügen','Deine Pläne','Wissenscheck','Auswertung','Deine Antwort'][page]}</Text>
 <ScrollView contentContainerStyle={{flexGrow:1}}>
 {page===0&&<View className="gap-3"><SelectionOptionRow Icon={Pencil} label="Klausur" selected={false} onPress={()=>{}}/><SelectionOptionRow Icon={Pencil} label="Vokabeltest" selected onPress={()=>{}}/><SelectionOptionRow Icon={Pencil} label="Italienisch" selected onPress={()=>{}}/></View>}
 {page===1&&<View><Text className="text-body-3 text-secondary-text">Deine Unterlagen bilden die Grundlage für deinen Lernplan.</Text><MaterialUploadActionCard hasSchoolMaterial canUpload onPress={()=>{}}/><Text className="mt-7 mb-3 text-body-4 text-secondary-text">Hochgeladen</Text><MaterialCard name="dayova_italienisch_vokabeln.pdf" size={297677} onRemove={()=>{}}/></View>}
 {page===2&&<LearningPlanCardVisual accessibilityLabel="Lernplan Italienisch" accessibilityHint="Lernplan öffnen" onPress={()=>{}} model={{subject:'Italienisch',status:{label:'Geplant',background:'#F1F7FB',foreground:'#00BAFF'},examDateLabel:'Freitag, 30. Oktober',currentTitle:'Wissenscheck · 5–10 Fragen',state:{kind:'ready',durationMinutes:10,progress:0,remainingDays:28,rollingWindowLabel:'0 abgeschlossen · 2 weitere Termine geplant'}}}/>}
 {page===3&&<View><Text className="font-semibold text-body-1 text-text">Welcher Satz drückt korrekt aus, dass man Äpfel (le mele) mag?</Text><ChoiceList item={{id:'preview',choices:[{id:'a',text:'Mi piace le mele.'},{id:'b',text:'Mi piacciono le mele.'},{id:'c',text:'Mi piacciono la mela.'}]}} selectedChoiceId={selected} onSelect={setSelected} disabled={false}/></View>}
 {page===4&&<FeedbackView attempt={{rating:'correct',feedback:'Deine Auswahl ist richtig. Wenn das, was man mag, im Plural steht (le mele), muss auch das Verb im Plural stehen (piacciono).',perfectAnswer:'Mi piacciono le mele.'} as any}/>}
 {page===5&&<View className="flex-1"><Text className="font-semibold text-body-1 text-text">Schreibe die Zahl 28 als italienisches Wort.</Text><TextAnswer value={value} onChange={setValue} editable placeholder="Schreibe hier deine Antwort." fillAvailableSpace autoFocus={false}/></View>}
 </ScrollView>
 <Button className="mt-6" onPress={()=>setPage((page+1)%6)}><Text>Weiter</Text></Button>
 </View>;
}
function App(){return <DayovaThemeProvider><Preview/></DayovaThemeProvider>}
registerRootComponent(App);
